function ok_(data, message) { return { success: true, message: message || '', data: data === undefined ? null : data }; }
function fail_(message, code) { return { success: false, message: message || 'Terjadi kesalahan.', code: code || 'ERROR', data: null }; }

function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }

  var sh = ss_().getSheetByName(name);

}
function tz_() { return Session.getScriptTimeZone() || 'Asia/Jakarta'; }
function fmtDate_(d, p) { return Utilities.formatDate(d, tz_(), p); }

function clean_(v) { return String(v === null || v === undefined ? '' : v).trim().slice(0, 2000); }
function toNum_(v) { var n = Number(String(v === null || v === undefined ? '' : v).replace(/[^\d]/g, '')); return isNaN(n) ? 0 : n; }
function inArr_(arr, v) { return arr.indexOf(v) >= 0; }
function natCmp_(a, b) { return String(a).localeCompare(String(b), 'id', { numeric: true, sensitivity: 'base' }); }
function normAlamat_(a) { return String(a === null || a === undefined ? '' : a).trim().toUpperCase().replace(/\s+/g, ''); }
function isDate_(s) { return /^\d{4}-\d{2}-\d{2}$/.test(s) && !isNaN(new Date(s + 'T00:00:00').getTime()); }
function isPeriode_(s) { return /^\d{4}-(0[1-9]|1[0-2])$/.test(s); }
function isTime_(s) { return /^([01]\d|2[0-3]):[0-5]\d$/.test(s); }

function cell_(v, h) {
  if (v instanceof Date) {
    if (v.getFullYear() < 1950) return fmtDate_(v, 'HH:mm');
    if (h === 'Periode') return fmtDate_(v, 'yyyy-MM');
    if (h === 'DibuatPada' || h === 'TanggalVerifikasi' || h === 'Waktu') return fmtDate_(v, 'yyyy-MM-dd HH:mm:ss');
    return fmtDate_(v, 'yyyy-MM-dd');
  }
  if (NUM_COLS[h]) return Number(v) || 0;
  if (typeof v === 'string') {
    if (/^'[=+\-@]/.test(v)) return v.slice(1);   // buang apostrof pelindung formula
    return v;
  }
  return v === null || v === undefined ? '' : v;
}
function out_(v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'string' && /^[=+\-@]/.test(v)) return "'" + v;   // cegah injeksi formula
  return v;
}
function readAll_(name) {
  var sh = sheet_(name), last = sh.getLastRow(), cols = COLS[name];
  if (last < 2) return [];
  var vals = sh.getRange(2, 1, last - 1, cols.length).getValues(), out = [];
  for (var i = 0; i < vals.length; i++) {
    var r = vals[i];
    if (r.join('') === '') continue;
    var o = { _row: i + 2 };
    for (var c = 0; c < cols.length; c++) o[cols[c]] = cell_(r[c], cols[c]);
    out.push(o);
  }
  return out;
}
function insert_(name, obj) {
  var row = COLS[name].map(function (h) { return out_(obj[h]); });
  sheet_(name).appendRow(row);
  return obj;
}
function update_(name, rowNum, obj) {
  var cols = COLS[name], rng = sheet_(name).getRange(rowNum, 1, 1, cols.length), cur = rng.getValues()[0];
  var nv = cols.map(function (h, i) { return Object.prototype.hasOwnProperty.call(obj, h) ? out_(obj[h]) : cur[i]; });
  rng.setValues([nv]);
}
function removeRow_(name, rowNum) { sheet_(name).deleteRow(rowNum); }
function byId_(rows, id) { for (var i = 0; i < rows.length; i++) if (rows[i].ID === id) return rows[i]; return null; }
function strip_(o, keys) { var c = {}; for (var k in o) if (k !== '_row' && keys.indexOf(k) < 0) c[k] = o[k]; return c; }
function pickAll_(rows, hide) { return rows.map(function (r) { return strip_(r, hide || []); }); }

function settings_() {
  var o = {}, k;
  for (k in DEFAULT_SETTINGS) o[k] = DEFAULT_SETTINGS[k];
  readAll_(SH.SETTING).forEach(function (r) { o[r.Kunci] = r.Nilai; });
  return o;
}
function setSetting_(key, val) {
  var rows = readAll_(SH.SETTING), f = null;
  rows.forEach(function (r) { if (r.Kunci === key) f = r; });
  if (f) update_(SH.SETTING, f._row, { Nilai: val }); else insert_(SH.SETTING, { Kunci: key, Nilai: val });
}

function hash_(salt, pw) {
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + pw, Utilities.Charset.UTF_8);
  return d.map(function (b) { return ('0' + (b & 0xFF).toString(16)).slice(-2); }).join('');
}
function makePw_(pw) { var salt = Utilities.getUuid().replace(/-/g, '').slice(0, 16); return salt + '$' + hash_(salt, pw); }
function checkPw_(pw, stored) {
  if (!stored || String(stored).indexOf('$') < 0) return false;
  var parts = String(stored).split('$');
  return hash_(parts[0], pw) === parts[1];
}
function newSession_(data) {
  var token = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  CacheService.getScriptCache().put('S_' + token, JSON.stringify(data), APP.SESSION_SEC);
  return token;
}
function getSession_(token) {
  if (!token || typeof token !== 'string') return null;
  var cache = CacheService.getScriptCache(), raw = cache.get('S_' + token);
  if (!raw) return null;
  var s = JSON.parse(raw);
  if (s.type === 'pengurus') {
    var p = byId_(readAll_(SH.PENGURUS), s.id);
    if (!p || p.Status !== 'Aktif') { cache.remove('S_' + token); return null; }
    s.role = p.Role; s.nama = p.Nama; s.jabatan = p.Jabatan;
  }
  cache.put('S_' + token, JSON.stringify(s), APP.SESSION_SEC);
  return s;
}
function checkLock_(key) {
  var n = Number(CacheService.getScriptCache().get('F_' + key)) || 0;
  if (n >= APP.MAX_FAIL) throw new Error('Terlalu banyak percobaan gagal. Coba lagi dalam 15 menit.');
}
function addFail_(key) {
  var c = CacheService.getScriptCache(), n = (Number(c.get('F_' + key)) || 0) + 1;
  c.put('F_' + key, String(n), APP.LOCK_SEC);
}
function clearFail_(key) { CacheService.getScriptCache().remove('F_' + key); }

function log_(user, aksi, detail) {
  try { insert_(SH.LOG, { Waktu: nowStr_(), User: user, Aksi: aksi, Detail: String(detail || '').slice(0, 500) }); } catch (e) { console.error(e); }
}

function perm_(role) {
  return {
    pengurus: role !== 'Warga',
    ubahData: inArr_(DATA_ROLES, role),        // rumah, warga, agenda, informasi
    ubahKeuangan: inArr_(KEU_ROLES, role),     // kas, verifikasi, tarif, catat bayar
    admin: role === 'Admin'
  };
}
function userView_(s) {
  return { type: s.type, id: s.id || '', nama: s.nama, role: s.role, jabatan: s.jabatan || '', alamat: s.alamat || '', perm: perm_(s.role) };
}

/** Pembungkus: validasi sesi + role + kunci tulis + tangkap error. */
function run_(token, roles, write, fn) {
  var lock = null;
  try {
    var s = getSession_(token);
    if (!s) return fail_('Sesi berakhir. Silakan login kembali.', 'SESSION');
    if (roles && roles.indexOf(s.role) < 0) return fail_('Anda tidak memiliki hak akses untuk fitur ini.', 'FORBIDDEN');
    if (write) { lock = LockService.getScriptLock(); lock.waitLock(20000); }
    return fn(s);
  } catch (e) {
    console.error(e && e.stack ? e.stack : e);
    return fail_(e && e.message ? e.message : String(e));
  } finally {
    if (lock) { try { lock.releaseLock(); } catch (x) {} }
  }
}

function apiGetPublicInfo() {
  try {
    var st = settings_();
    var alamat = readAll_(SH.RUMAH).map(function (r) { return r.Alamat; }).sort(natCmp_);
    return ok_({ namaAplikasi: st.NamaAplikasi, namaRT: st.NamaRT, alamatRT: st.AlamatRT, pinWargaAktif: st.PinWargaAktif === 'YA', daftarAlamat: alamat });
  } catch (e) { return fail_(e.message); }
}

function apiLoginWarga(alamat, pin) {
  try {
    var a = normAlamat_(alamat);
    if (!a) throw new Error('Pilih atau isi alamat rumah.');
    checkLock_('W' + a);
    var rumah = null;
    readAll_(SH.RUMAH).forEach(function (r) { if (normAlamat_(r.Alamat) === a) rumah = r; });
    if (!rumah) { addFail_('W' + a); throw new Error('Alamat rumah tidak ditemukan.'); }
    if (settings_().PinWargaAktif === 'YA') {
      if (!rumah.PINHash) throw new Error('PIN rumah ini belum dibuat. Hubungi pengurus RT.');
      if (!checkPw_(String(pin || ''), rumah.PINHash)) { addFail_('W' + a); throw new Error('PIN salah.'); }
    }
    clearFail_('W' + a);
    var sess = { type: 'warga', role: 'Warga', alamat: rumah.Alamat, nama: 'Rumah ' + rumah.Alamat };
    var token = newSession_(sess);
    log_(sess.nama, 'LOGIN', 'Login warga');
    return ok_({ token: token, user: userView_(sess) }, 'Login berhasil');
  } catch (e) { return fail_(e.message); }
}

function apiLoginPengurus(username, password) {
  try {
    var u = clean_(username).toLowerCase();
    if (!u || !password) throw new Error('Username dan password wajib diisi.');
    checkLock_('P' + u);
    var p = null;
    readAll_(SH.PENGURUS).forEach(function (x) { if (String(x.Username).toLowerCase() === u) p = x; });
    if (!p || !checkPw_(String(password), p.PasswordHash)) { addFail_('P' + u); throw new Error('Username atau password salah.'); }
    if (p.Status !== 'Aktif') throw new Error('Akun dinonaktifkan. Hubungi Admin.');
    clearFail_('P' + u);
    var sess = { type: 'pengurus', id: p.ID, role: p.Role, nama: p.Nama, jabatan: p.Jabatan };
    var token = newSession_(sess);
    log_(p.Nama, 'LOGIN', 'Login pengurus (' + p.Role + ')');
    return ok_({ token: token, user: userView_(sess) }, 'Login berhasil');
  } catch (e) { return fail_(e.message); }
}

function apiCekSesi(token) {
  try {
    var s = getSession_(token);
    if (!s) return fail_('Sesi berakhir.', 'SESSION');
    return ok_({ user: userView_(s) });
  } catch (e) { return fail_(e.message); }
}

function apiLogout(token) {
  try {
    if (token) { var s = getSession_(token); if (s) log_(s.nama, 'LOGOUT', ''); CacheService.getScriptCache().remove('S_' + token); }
    return ok_(null, 'Anda telah keluar.');
  } catch (e) { return fail_(e.message); }
}

function findRumah_(alamat) {
  var a = normAlamat_(alamat), f = null;
  readAll_(SH.RUMAH).forEach(function (r) { if (normAlamat_(r.Alamat) === a) f = r; });
  return f;
}
function tarifFor_(tarifRows, jenis, kategori) {
  var exact = 0, umum = 0;
  tarifRows.forEach(function (t) {
    if (t.Status !== 'Aktif' || t.JenisIuran !== jenis) return;
    if (t.KategoriIuran === kategori) exact = t.Nominal;
    if (t.KategoriIuran === 'Umum') umum = t.Nominal;
  });
  return exact || umum || 0;
}
function jenisAktif_(tarifRows) {
  var seen = {}, out = [];
  tarifRows.forEach(function (t) { if (t.Status === 'Aktif' && !seen[t.JenisIuran]) { seen[t.JenisIuran] = 1; out.push(t.JenisIuran); } });
  return out;
}
function payIndex_(pays) {
  var idx = {};
  pays.forEach(function (p) {
    if (p.Status === 'Ditolak') return;
    var k = p.AlamatRumah + '|' + p.JenisIuran + '|' + p.Periode;
    if (!idx[k] || p.Status === 'Terverifikasi') idx[k] = p.Status;
  });
  return idx;
}
function periodeList_(start, end) {
  var out = [], y = Number(start.slice(0, 4)), m = Number(start.slice(5, 7));
  while (out.length < 600) {
    var p = y + '-' + ('0' + m).slice(-2);
    if (p > end) break;
    out.push(p); m++; if (m > 12) { m = 1; y++; }
  }
  return out;
}
function startPeriode_(st) { return isPeriode_(st.MulaiIuran) ? st.MulaiIuran : (new Date().getFullYear() + '-01'); }

/** Tunggakan satu rumah sampai bulan berjalan. Status "menunggu verifikasi" tidak dihitung tunggakan. */
function tunggakanRumah_(r, idx, tarif, start, end) {
  var items = [], pers = periodeList_(start, end);
  jenisAktif_(tarif).forEach(function (j) {
    var nom = tarifFor_(tarif, j, r.KategoriIuran || 'Umum');
    if (!nom) return;
    var miss = pers.filter(function (p) { return !idx[r.Alamat + '|' + j + '|' + p]; });
    if (miss.length) items.push({ jenis: j, nominal: nom, periode: miss, jumlah: miss.length, subtotal: nom * miss.length });
  });
  return items;
}
function groupPay_(rows) {
  var m = {}, order = [];
  rows.forEach(function (p) {
    var k = p.KodeTransaksi || p.ID;
    if (!m[k]) {
      m[k] = { kode: k, alamat: p.AlamatRumah, jenis: p.JenisIuran, periode: [], ids: [], total: 0, tanggalBayar: p.TanggalBayar,
               metode: p.MetodeBayar, adaBukti: !!p.BuktiURL, status: p.Status, catatan: p.CatatanVerifikator,
               diverifikasiOleh: p.DiverifikasiOleh, tanggalVerifikasi: p.TanggalVerifikasi };
      order.push(k);
    }
    m[k].periode.push(p.Periode); m[k].ids.push(p.ID); m[k].total += p.Nominal;
  });
  return order.map(function (k) { m[k].periode.sort(); return m[k]; });
}

    Nominal: p.Nominal, SumberReferensi: p.ID, DicatatOleh: user
  });
}

function kasSummary_(kas) {
  var nowP = periodeNow_(), saldo = 0, tin = 0, tout = 0, bin = 0, bout = 0, map = {};
  kas.forEach(function (k) {
    var p = String(k.Tanggal).slice(0, 7); if (!map[p]) map[p] = { masuk: 0, keluar: 0 };
    if (k.Jenis === 'Pemasukan') { saldo += k.Nominal; tin += k.Nominal; map[p].masuk += k.Nominal; if (p === nowP) bin += k.Nominal; }
    else { saldo -= k.Nominal; tout += k.Nominal; map[p].keluar += k.Nominal; if (p === nowP) bout += k.Nominal; }
  });
  var series = [], now = new Date();
  for (var i = 5; i >= 0; i--) {
    var d = new Date(now.getFullYear(), now.getMonth() - i, 1), p2 = fmtDate_(d, 'yyyy-MM');
    series.push({ periode: p2, label: BULAN[d.getMonth()].slice(0, 3) + ' ' + String(d.getFullYear()).slice(2),
                  masuk: (map[p2] || {}).masuk || 0, keluar: (map[p2] || {}).keluar || 0 });
  }
  return { saldo: saldo, totalMasuk: tin, totalKeluar: tout, bulanMasuk: bin, bulanKeluar: bout, series: series };
}
function agendaView_(a) {
  var o = strip_(a, []);
  o.StatusTampil = (a.Status === 'Akan Datang' && a.Tanggal < today_()) ? 'Selesai' : a.Status;
  return o;
}

function apiGetDashboard(token) {
  return run_(token, ALL_ROLES, false, function (s) {
    var st = settings_(), kas = readAll_(SH.KAS), sum = kasSummary_(kas);
    var agenda = readAll_(SH.AGENDA).map(agendaView_).filter(function (a) { return a.StatusTampil === 'Akan Datang'; })
      .sort(function (a, b) { return (a.Tanggal + (a.Waktu || '')).localeCompare(b.Tanggal + (b.Waktu || '')); }).slice(0, 4);
    var info = readAll_(SH.INFO).filter(function (i) { return i.Status === 'Publish'; })
      .sort(function (a, b) { return String(b.TanggalTerbit).localeCompare(String(a.TanggalTerbit)); }).slice(0, 4).map(infoView_);
    var tarif = readAll_(SH.TARIF), pays = readAll_(SH.PEMBAYARAN), idx = payIndex_(pays);
    var start = startPeriode_(st), end = periodeNow_();
    var data = { role: s.role, namaRT: st.NamaRT, saldo: sum.saldo, bulanIni: { masuk: sum.bulanMasuk, keluar: sum.bulanKeluar },
                 series: sum.series, agenda: agenda, info: info };

    if (s.type === 'warga') {
      var r = findRumah_(s.alamat);
      if (!r) throw new Error('Data rumah tidak ditemukan.');
      var tg = tunggakanRumah_(r, idx, tarif, start, end), tb = 0, tn = 0;
      tg.forEach(function (i) { tb += i.jumlah; tn += i.subtotal; });
      var menunggu = groupPay_(pays.filter(function (p) { return p.AlamatRumah === r.Alamat && p.Status === 'Menunggu Verifikasi'; })).length;
      data.rumah = { alamat: r.Alamat, statusHunian: r.StatusHunian };
      data.iuran = { lunas: tg.length === 0, tunggakan: tg, totalBulan: tb, totalNominal: tn, menunggu: menunggu };
    } else {
      var rumah = readAll_(SH.RUMAH), warga = readAll_(SH.WARGA), per = {};
      STATUS_HUNIAN.forEach(function (x) { per[x] = 0; });
      rumah.forEach(function (r2) { per[r2.StatusHunian] = (per[r2.StatusHunian] || 0) + 1; });
      var menunggak = 0, totalTg = 0;
      rumah.forEach(function (r2) {
        var t = tunggakanRumah_(r2, idx, tarif, start, end);
        if (t.length) { menunggak++; t.forEach(function (i) { totalTg += i.subtotal; }); }
      });
      var tinggal = warga.filter(function (w) { return w.StatusTinggal === 'Tinggal'; }).length;
      data.ringkas = {
        totalRumah: rumah.length, totalWarga: warga.length, wargaTinggal: tinggal, perStatus: per,
        menungguVerifikasi: groupPay_(pays.filter(function (p) { return p.Status === 'Menunggu Verifikasi'; })).length,
        rumahMenunggak: menunggak, totalTunggakan: totalTg, totalPemasukan: sum.totalMasuk, totalPengeluaran: sum.totalKeluar
      };
    }
    return ok_(data);
  });
}

function apiSaveRumah(token, obj) {
  return run_(token, DATA_ROLES, true, function (s) {
    obj = obj || {};
    var alamat = normAlamat_(obj.Alamat);
    if (!alamat) throw new Error('Alamat rumah wajib diisi.');
    if (!/^[A-Z0-9\-\.]+(\/[A-Z0-9\-\.]+)?$/.test(alamat)) throw new Error('Format alamat tidak valid (contoh: A4/1).');
    var status = clean_(obj.StatusHunian);
    if (!inArr_(STATUS_HUNIAN, status)) throw new Error('Status hunian tidak valid.');
    var pemilik = clean_(obj.NamaPemilik), penghuni = clean_(obj.NamaPenghuni);
    if (status === 'Dihuni Pemilik') { if (!pemilik) throw new Error('Nama pemilik wajib diisi.'); penghuni = pemilik; }
    if (status === 'Dikontrak' && !penghuni) throw new Error('Nama penghuni (pengontrak) wajib diisi.');
    if (status === 'Kosong') { if (!pemilik) throw new Error('Nama pemilik wajib diisi.'); penghuni = ''; }
    var rows = readAll_(SH.RUMAH);
    var dup = rows.filter(function (r) { return normAlamat_(r.Alamat) === alamat && r.ID !== obj.ID; });
    if (dup.length) throw new Error('Alamat ' + alamat + ' sudah terdaftar.');
    var parts = alamat.split('/');
    var data = {
      Alamat: alamat, Blok: parts[0], Nomor: parts[1] || '', StatusHunian: status, NamaPemilik: pemilik, NamaPenghuni: penghuni,
      NoHP: clean_(obj.NoHP).replace(/[^\d+]/g, ''), KategoriIuran: clean_(obj.KategoriIuran) || 'Umum', Keterangan: clean_(obj.Keterangan)
    };
    if (obj.ID) {
      var ex = byId_(rows, obj.ID); if (!ex) throw new Error('Data rumah tidak ditemukan.');
      if (ex.Alamat !== alamat) throw new Error('Alamat tidak dapat diubah. Hapus lalu buat rumah baru bila salah penulisan.');
      update_(SH.RUMAH, ex._row, data);
      // rumah kosong -> semua anggota "Tidak Tinggal"
      if (status === 'Kosong') {
        readAll_(SH.WARGA).forEach(function (w) { if (w.AlamatRumah === alamat && w.StatusTinggal !== 'Tidak Tinggal') update_(SH.WARGA, w._row, { StatusTinggal: 'Tidak Tinggal' }); });
      }
      recalcRumah_(alamat);
      log_(s.nama, 'UBAH_RUMAH', alamat);
      return ok_({ ID: ex.ID }, 'Data rumah diperbarui.');
    }
    data.ID = genId_('RMH'); data.NamaKepalaKeluarga = ''; data.JumlahJiwa = 0; data.PINHash = '';
    insert_(SH.RUMAH, data);
    log_(s.nama, 'TAMBAH_RUMAH', alamat);
    return ok_({ ID: data.ID }, 'Rumah ' + alamat + ' ditambahkan.');
  });
}

function getFolder_(name, settingKey) {
  if (settingKey) {
    var id = settings_()[settingKey];
    if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  }
  var it = DriveApp.getFoldersByName(name), f = it.hasNext() ? it.next() : DriveApp.createFolder(name);
  if (settingKey) setSetting_(settingKey, f.getId());
  return f;
}

function saveBukti_(p, kode, alamat) {
  var mime = clean_(p.mimeType).toLowerCase();
  var ext = { 'image/jpeg': '.jpg', 'image/png': '.png', 'application/pdf': '.pdf' }[mime];
  if (!ext) throw new Error('Tipe file tidak didukung. Gunakan JPG, PNG, atau PDF.');
  var b64 = String(p.base64 || '').replace(/^data:[^;]+;base64,/, '');
  if (!b64) throw new Error('File bukti kosong.');
  if (Math.floor(b64.length * 3 / 4) > APP.MAX_FILE_BYTES) throw new Error('Ukuran file bukti maksimal 2 MB.');
  var name = kode + '_' + alamat.replace(/[^A-Z0-9]/gi, '-') + ext;
  var blob = Utilities.newBlob(Utilities.base64Decode(b64), mime, name);
  var file = getFolder_(APP.FOLDER_BUKTI, 'FolderBuktiId').createFile(blob);
  return file.getUrl();
}

function createPayments_(s, alamat, p, auto) {
  p = p || {};
  var rumah = findRumah_(alamat); if (!rumah) throw new Error('Rumah tidak ditemukan.');
  var jenis = clean_(p.jenis); if (!jenis) throw new Error('Pilih jenis iuran.');
  var tarif = readAll_(SH.TARIF);
  var nominal = tarifFor_(tarif, jenis, rumah.KategoriIuran || 'Umum');
  if (!nominal) throw new Error('Tarif iuran "' + jenis + '" belum diatur.');
  var seen = {}, periode = [];
  (Array.isArray(p.periode) ? p.periode : []).forEach(function (x) { x = clean_(x); if (!seen[x]) { seen[x] = 1; periode.push(x); } });
  if (!periode.length) throw new Error('Pilih minimal satu bulan.');
  if (periode.length > 24) throw new Error('Maksimal 24 bulan dalam satu pembayaran.');
  var maxYear = new Date().getFullYear() + 2;
  periode.forEach(function (x) {
    if (!isPeriode_(x)) throw new Error('Periode tidak valid: ' + x);
    if (Number(x.slice(0, 4)) > maxYear || Number(x.slice(0, 4)) < 2000) throw new Error('Periode di luar rentang: ' + x);
  });
  var tgl = clean_(p.tanggalBayar) || today_(); if (!isDate_(tgl)) throw new Error('Tanggal bayar tidak valid.');
  var metode = clean_(p.metode) || 'Transfer';
  var idx = payIndex_(readAll_(SH.PEMBAYARAN));
  var dup = periode.filter(function (x) { return idx[rumah.Alamat + '|' + jenis + '|' + x]; });
  if (dup.length) throw new Error('Periode berikut sudah dibayar atau sedang menunggu verifikasi: ' + dup.map(periodeLabel_).join(', '));
  var kode = genKode_(), url = '';
  if (p.base64) url = saveBukti_(p, kode, rumah.Alamat);
  else if (!auto) throw new Error('Bukti pembayaran wajib diunggah.');
  periode.sort();
  periode.forEach(function (per) {
    var row = { ID: genId_('PAY'), AlamatRumah: rumah.Alamat, JenisIuran: jenis, Periode: per, Nominal: nominal, TanggalBayar: tgl, MetodeBayar: metode,
                BuktiURL: url, KodeTransaksi: kode, Status: auto ? 'Terverifikasi' : 'Menunggu Verifikasi',
                CatatanVerifikator: auto ? clean_(p.catatan) : '', DiverifikasiOleh: auto ? s.nama : '', TanggalVerifikasi: auto ? nowStr_() : '' };
    insert_(SH.PEMBAYARAN, row);
    if (auto) addKasFromPayment_(row, s.nama);
  });
  log_(s.nama, auto ? 'CATAT_BAYAR' : 'UPLOAD_BAYAR', rumah.Alamat + ' ' + jenis + ' ' + periode.join(',') + ' ' + rp_(nominal * periode.length));
  return { kode: kode, jumlahBulan: periode.length, total: nominal * periode.length, status: auto ? 'Terverifikasi' : 'Menunggu Verifikasi' };
}

/** Ambil bukti bayar sebagai data URL (tanpa membuka akses publik file Drive). */

/** Riwayat iuran per rumah (warga: rumahnya sendiri; pengurus: parameter alamat). */
function apiGetRiwayatIuran(token, tahun, alamat) {
  return run_(token, ALL_ROLES, false, function (s) {
    var r = findRumah_(resolveAlamat_(s, alamat));
    if (!r) throw new Error('Rumah tidak ditemukan.');
    var st = settings_(), tarif = readAll_(SH.TARIF), pays = readAll_(SH.PEMBAYARAN).filter(function (p) { return p.AlamatRumah === r.Alamat; });
    var th = Number(tahun) || new Date().getFullYear(), start = startPeriode_(st), nowP = periodeNow_(), jenis = jenisAktif_(tarif);
    pays.forEach(function (p) { if (jenis.indexOf(p.JenisIuran) < 0) jenis.push(p.JenisIuran); });
    var idx = payIndex_(pays), grid = {};
    jenis.forEach(function (j) {
      grid[j] = {};
      for (var m = 1; m <= 12; m++) {
        var mm = ('0' + m).slice(-2), per = th + '-' + mm, stt = idx[r.Alamat + '|' + j + '|' + per];
        var v = stt === 'Terverifikasi' ? 'lunas' : stt ? 'menunggu' : per < start ? 'na' : per > nowP ? 'akan' : 'belum';
        grid[j][mm] = v;
      }
    });
    var tg = tunggakanRumah_(r, idx, tarif, start, nowP), tn = 0;
    tg.forEach(function (i) { tn += i.subtotal; });
    var trx = groupPay_(pays).sort(function (a, b) { return String(b.tanggalBayar).localeCompare(String(a.tanggalBayar)); });
    return ok_({ alamat: r.Alamat, statusHunian: r.StatusHunian, tahun: th, jenis: jenis, grid: grid, transaksi: trx, tunggakan: tg, totalTunggakan: tn });
  });
}

    var user = clean_(obj.Username).toLowerCase();

    baru = String(baru || '');
    if (baru.length < 6) throw new Error('Password baru minimal 6 karakter.');
    if (baru === lama) throw new Error('Password baru harus berbeda dari password lama.');
    update_(SH.PENGURUS, p._row, { PasswordHash: makePw_(baru) });
    log_(s.nama, 'UBAH_PASSWORD', p.Username);
    return ok_(null, 'Password berhasil diubah.');
  });
}

function apiGetPengaturan(token) {
  return run_(token, PENGURUS_ROLES, false, function () {
    var st = settings_(), o = {};
    EDITABLE_SETTINGS.forEach(function (k) { o[k] = st[k]; });
    return ok_(o);
  });
}

function apiSavePengaturan(token, obj) {
  return run_(token, ADMIN_ROLES, true, function (s) {
    obj = obj || {};
    EDITABLE_SETTINGS.forEach(function (k) {
      if (!Object.prototype.hasOwnProperty.call(obj, k)) return;
      var v = clean_(obj[k]);
      if (k === 'PinWargaAktif' && v !== 'YA' && v !== 'TIDAK') throw new Error('PinWargaAktif harus YA atau TIDAK.');
      if (k === 'MulaiIuran' && !isPeriode_(v)) throw new Error('Mulai iuran harus berformat yyyy-mm.');
      if ((k === 'NamaAplikasi' || k === 'NamaRT') && !v) throw new Error('Nama aplikasi/RT wajib diisi.');
      setSetting_(k, v);
    });
    log_(s.nama, 'UBAH_PENGATURAN', Object.keys(obj).join(','));
    return ok_(null, 'Pengaturan disimpan.');
  });
}

function ensureSheet_(name) {
  var ss = ss_(), sh = ss.getSheetByName(name) || ss.insertSheet(name), cols = COLS[name];
  if (sh.getLastRow() === 0 || String(sh.getRange(1, 1).getValue()) === '') {
    sh.getRange(1, 1, 1, cols.length).setValues([cols])
      .setFontWeight('bold').setBackground('#1e6f5c').setFontColor('#ffffff');
    sh.setFrozenRows(1);
    // format teks agar tanggal/periode tidak diubah Sheets menjadi tipe Date
    cols.forEach(function (h, i) {
      if (!NUM_COLS[h]) sh.getRange(2, i + 1, Math.max(sh.getMaxRows() - 1, 1), 1).setNumberFormat('@');
    });
  } else {
    var cur = sh.getRange(1, 1, 1, cols.length).getValues()[0].join('|');
    if (cur !== cols.join('|')) Logger.log('PERINGATAN: header sheet "' + name + '" berbeda dari definisi. Periksa kolom.');
  }
  return sh;
}
